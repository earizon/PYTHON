  ```
  |               ·  ·            #       └──────┬──────┘ 
  |               ·  ·            #       stack of "outer" function while 
  |               ·  ·            #       building the lambda => inmutable state 
  |               ·  ·            #       once built.
  |               └──┴─············· func. args

  | (lambda  x, y: x + y)(5,  3) # 8
  |
  |  def build_adder(n):          # <·· function building(returning) new lambdas
  |      return lambda x: x + n   #     Each lambda keeps in stack the different n
  |  plus_3 = build_adder(3)      #     of each invocation to the outer function
  |
  | # -- (args) vs *(kwargs) ---------------------------------------------------
  |
  | - In function declaration:
  |   - '*'  means "pack all remaining positional arguments into
  |                 a tuple named <name>"
  |   - '**' means "pack all remaining positional arguments into
  |                 a dictionary named <name>"
  | 
  | - In funct. call:
  |   - '*'  means "unpack tuple or list named <name> 
  |                to positional arguments at this position"
  |   - '**' mean: "unpack dictionary    named <name> 
  |                to positional arguments at this position"
  | 
  | def forward(f,  args,  *kwargs): # <·· Ex. wrapper func. around any other function
  |     return f( args,  *kwargs)
  ```

  ```
  | 'Alice' 'Bob'            # <·· STRING CONCATENATION ('AliceBob')

  | 'Hello Mr {} {} """      # <·· STRING FORMAT (2.7+) {}, ...:

  | # -- Escape chars ----------------------------------------------------------
  | Escape char   Prints as ..
  | ---------     ------------
  | \'            Single quote
  | \"            Double quote
  | \t            Tab
  | \n            Newline
  | \\            Backslash
  |
  |       ┌·· Raw Strings : ignores all escape characters (used for regex mostly)

  | # ----- multi line string  ------

  | str01. endswith ('world!')  # True
  | str01[0]     # 'H'        
  | str01[0:5]   # 'Hello'
  | str01[::-1]  # '!dlrow olleH'
  | 'Hello' in str01  # True    
  | 'HELLO' in str01  # False   
  | str01.upper()   # 'HELLO WORLD!'

  | str01.isupper() # False
  | str01.upper().isupper()  # True
  ```
