# Regex 

* <https://docs.python.org/3.6/library/re.html>
* <https://developers.google.com/edu/python/regular-expressions>


  ```
  | import  re                                 # Step 1
  │ # ───────────────────────────────── GROUPING WITH PARENTHESES
  | INPUT = 'My number is 415-555-4242.'
  | REGEX_EXP=r'(\d\d\d)-(\d\d\d-\d\d\d\d)'    # r => "raw" string. Note, expression can be split
  |                                            # into N lines,  also adding comments like:
  |                                            # (Don't forget to add the re.VERBOSE flag when compiling)
  |                                            # r'''(
  |                                            #     (\d{3}|\(\d{3}\))?            # area code
  |                                            #     (\s|-|\.)?                    # separator
  |                                            #     \d{3}                         # first 3 digits
  |                                            #     (\s|-|\.)                     # separator
  |                                            #     \d{4}                         # last 4 digits
  |                                            #     (\s*(ext|x|ext.)\s*\d{2,5})?  # extension
  |                                            # )'''
  |                                            # ... re.compile(..., re.VERBOSE | re.IGNORECASE)   
  |                                            # Don't forget!!! ────┴────────┘
  |
  | REGEX = re.compile(REGEX_EXP)              # step 2. Create regex object
  | #       re.compile(REGEX_EXP, re.DOTALL)   # <·· re.DOTALL flag will match after new lines
  | #                                          # <·· re.IGNORECASE ignores case
  | # search  ──────────────────────────────────────────────────────────────────
  | mo = REGEX.search(INPUT)                   # step 3. Returns a Match object.
  |                                            # Use findall() instead to return a
  |                                            # searched string.
  | mo.group(1)                                # '415'
  | mo.group(2)                                # '555-4242'
  | mo.group(0)                                # '415-555-4242'
  | mo.group()                                 # '415-555-4242'
  | mo.groups()                                # ('415', '555-4242')
  | 
  | # greedy vs non-greedy ─────────────────────────────────────────────────────
  | INPUT='HaHaHaHaHa'
  | mo1 = re.compile(r'(Ha){3,5}' ).search(INPUT) # mo1.group() === 'HaHaHaHaHa' # greedy by default
  | mo2 = re.compile(r'(Ha){3,5}?').search(INPUT) # mo1.group() === 'HaHaHa'     # ? makes non-greedy
  | 
  | # findall ──────────────────────────────────────────────────────────────────
  | INPUT='Cell: 415-555-9999 Work: 212-555-0000'
  | REGEX=r'\d\d\d-\d\d\d-\d\d\d\d'
  | REGEX = re.compile(REGEX) # has no groups
  | REGEX.findall(INPUT)           # <·· ['415-555-9999', '212-555-0000'] (or None)
  |                                # when not using groups findall returns a list of matches
  |                                # When     using groups findall returns a list of strings
  |                                #          for each group such as:
  |                                #          [ ('415', '555', '9999'),
  |                                #            ('212', '555', '0000') ]
  | 
  | # character classes ────────────────────────────────────────────────────────
  | re.compile(r'[aeiou]').findall('Robocop eats') # ['o', 'o', 'o', 'e', 'a']
  | # or      (r'[a-zA-Z0-9]')   (range of included letters/numbers)
  | # or      (r'[^a-zA-Z0-9]')  (range of excluded letters/numbers)
  │ #             └─ ^ neglects inside [...], otherwise it means 'start of line or text'
  | #                $                        alwasy       means 'end of line or text' 
  | #                Example: r'^Hello$' <·· full line matches Hello
  | # 
  | # . : Match any character  | \.: Match '.'
  | # \d: Match any digit.     | \w: Match any word
  | # \s: Match any space (tab, white-space, ...)
  |
  | # (.*) : Match any number of characters (greedy, largest match)
  | # (.*?): Match any number of characters (non-greedy)
  |
  | Symbol  Matches
  | ?       zero or one of the preceding group.
  | *       zero or more of the preceding group.
  | +       one or more of the preceding group.
  | {n}     exactly n of the preceding group.
  | {n,}    n or more of the preceding group.
  | {,m}    0 to m of the preceding group.
  | {n,m}   at least n and at most m of the preceding p.
  | {n,m}? *? +?  performs a nongreedy match of the preceding p. 
  | .       any character, except newline \n.
  | \d, \w, and \s                  a digit, word, or space character, respectively.
  | \D, \W, and \S  anything except a digit, word, or space character, respectively.
  | 
  | # sub(tituting strings) ────────────────────────────────────────────────────
  | INPUT = 'Alice gave the secret documents to Bob'
  | names_regex = re.compile(r'(Alice|Bob|secret documents)')
  | names_regex.sub('XXX', INPUT) # 'XXX gave the XXX to XXX.'
  ```
